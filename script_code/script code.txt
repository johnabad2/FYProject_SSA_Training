"""
run_tests.py - one place to run every SSA test from inside Blender.

HOW TO USE
  1. Change TEST below to one of the names in the list.
  2. Press  Run Script  (Alt+P) with the mouse over this text.
  3. Watch the System Console (Window > Toggle System Console) for progress.

TESTS
  "setup"      black space world, Sun, camera, satellites laid out            (instant)
  "stills"     6 still renders of both satellites at different angles         (~1 min)
  "dawn_dusk"  10 frames of the dawn-dusk pass for both satellites            (~1 min)
  "animation"  render the 240 animation frames (resumable) and, when all
               exist, assemble dawn_dusk_pass.mp4                              (~10+ min)
  "materials"  flat plate / matte sphere / shiny panel spin sweeps + distance (~5 min)
  "fidelity"   detailed vs cheap model brightness comparison                  (~15 min)

Blender is busy (frozen) while a test runs; that is normal. Re-running an unfinished
"animation" or "fidelity" continues where it left off.
"""
import os
import runpy

HERE = r"C:\SSA\FYProject_SSA_Training"
TEST = "setup"                  # <-- change this


def load(name):
    return runpy.run_path(os.path.join(HERE, name), run_name="ssa_" + name)


def main(test=TEST):
    if test == "setup":
        return load("setup_scene.py")["setup"]()
    if test == "stills":
        return load("render_angles.py")["main"]()
    if test == "dawn_dusk":
        return load("dawn_dusk_scenario.py")["run"]()
    if test == "animation":
        ns = load("make_animation.py")
        res = ns["render_frames"](max_seconds=100)
        print(res)
        return ns["assemble"]() if res["remaining"] == 0 else res
    if test == "materials":
        ns = load("material_tests.py")
        out = [ns["spin_sweep"](k) for k in ("plate_diffuse", "sphere_diffuse", "panel_shiny")]
        out.append(ns["distance_sweep"]())
        return out
    if test == "fidelity":
        return load("fidelity_test.py")["run"]()
    raise ValueError(f"unknown TEST {test!r}")


print(main())

"""
setup_scene.py - put the open Blender scene into the state the SSA tests expect.

  * pure black "space" world (no ambient light)
  * one Sun lamp (RenderSun) with the Sun's real angular size, strength 4
  * one camera (RenderCam) framed on Aqua, set as the active camera
  * Aqua at the origin, Suomi NPP 50 units away, so the two satellites do not overlap
  * EEVEE, 1280 x 720, 64 samples; the 3D view shows the camera view in Rendered mode
  * any other lights are switched off (hidden, not deleted)

Safe to run again at any time.
  ns = runpy.run_path(r"C:\\SSA\\FYProject_SSA_Training\\setup_scene.py", run_name="setup"); ns["setup"]()
"""
import math
import os
import runpy

import bpy

HERE = r"C:\SSA\FYProject_SSA_Training"
SUN_AZ, SUN_EL = 45, 35        # Sun direction seen from the satellite
CAM_AZ, CAM_EL = 45, 20        # camera direction seen from the satellite
RES = (1280, 720)
SAMPLES = 64
LAYOUT = {"Aqua_Detailed_pivot": (0, 0, 0), "SuomiNPP_LowPoly_pivot": (0, -50, 0)}


def setup(frame_model=0, look_through_camera=True):
    sc = bpy.context.scene
    ra = runpy.run_path(os.path.join(HERE, "render_angles.py"), run_name="render_angles")

    for name, loc in LAYOUT.items():
        if name in bpy.data.objects:
            bpy.data.objects[name].location = loc
    bpy.context.view_layer.update()

    ra["AMBIENT"] = 0.0
    world = ra["ensure_space_world"]()
    world.node_tree.nodes["Background"].inputs["Color"].default_value = (0, 0, 0, 1)
    world.node_tree.nodes["Background"].inputs["Strength"].default_value = 0.0
    sc.world = world

    sun, cam = ra["ensure_sun"](), ra["ensure_camera"]()
    for o in bpy.data.objects:                       # only the Sun lights the scene
        if o.type == 'LIGHT' and o is not sun:
            o.hide_render = o.hide_viewport = True

    sc.render.resolution_x, sc.render.resolution_y = RES
    sc.render.resolution_percentage = 100
    sc.render.engine = 'BLENDER_EEVEE'
    sc.eevee.taa_render_samples = SAMPLES
    sc.render.film_transparent = False

    g = ra["setup_group"](ra["MODELS"][frame_model])
    rx, ry = RES
    fov = 2 * math.atan(math.tan(cam.data.angle_x / 2) * min(rx, ry) / max(rx, ry))
    dist = g["radius"] * 1.05 / math.sin(fov / 2)
    centre = g["pivot"].matrix_world.translation.copy()
    sd, cd = ra["direction"](SUN_AZ, SUN_EL), ra["direction"](CAM_AZ, CAM_EL)
    sun.location = centre + sd * dist * 2
    ra["look_along"](sun, -sd)
    cam.location = centre + cd * dist
    ra["look_along"](cam, centre - cam.location)
    sc.camera = cam

    for area in bpy.context.screen.areas:
        if area.type == 'VIEW_3D':
            area.spaces[0].shading.type = 'RENDERED'
            if look_through_camera:
                area.spaces[0].region_3d.view_perspective = 'CAMERA'
    return dict(model=ra["MODELS"][frame_model]["label"], camera_distance=round(dist, 1))


if __name__ == "__main__":
    print(setup())


################################################################################
# FILE: render_angles.py
################################################################################

"""
render_angles.py - render the satellite models under different Sun angles,
viewing angles and satellite rotations.

Run it any of these ways:
  * Blender > Scripting tab > open this file > Run Script
  * Command line (no UI):
      blender -b satellite_comparison.blend -P render_angles.py -- --engine EEVEE --out C:\\SSA\\renders
  * From Python:  ns = runpy.run_path(path, run_name="render_angles"); ns["main"](...)

ANGLE CONVENTIONS (all in degrees)
  sun_az / cam_az : azimuth, measured counter-clockwise (seen from above, +Z up)
                    starting at the +X axis. 0 = +X, 90 = +Y, 180 = -X, 270 = -Y.
  sun_el / cam_el : elevation above the horizontal XY plane. 90 = straight above.
  rot_x / rot_z   : rotation of the satellite about its own centre (X then Z).
  The Sun / camera sit on a sphere around the satellite, so
  sun_az == cam_az and sun_el == cam_el means "Sun directly behind the camera"
  (the satellite is fully lit from the camera's point of view).

Outputs: one PNG per model per shot, plus render_log.csv with triangle counts
and render times (handy for the detailed vs low-poly comparison).
"""
import bpy
import csv
import math
import os
import sys
import time
from mathutils import Vector

# --------------------------------------------------------------------------
# CONFIG - edit these
# --------------------------------------------------------------------------
# collection / root-empty names created when the models were imported
MODELS = [
    dict(label="Aqua_detailed",    collection="Satellite_Detailed", root="Aqua_Detailed_root"),
    dict(label="SuomiNPP_lowpoly", collection="Satellite_LowPoly",  root="SuomiNPP_LowPoly_root"),
]

# A handful of shots. Add, remove or edit freely.
SHOTS = [
    dict(name="fully_lit",     sun_az=45,  sun_el=35, cam_az=45,  cam_el=20, rot_x=0, rot_z=0),
    dict(name="side_view",     sun_az=45,  sun_el=35, cam_az=135, cam_el=20, rot_x=0, rot_z=0),
    dict(name="rear_view",     sun_az=45,  sun_el=35, cam_az=225, cam_el=20, rot_x=0, rot_z=0),
    dict(name="rotated_90",    sun_az=45,  sun_el=35, cam_az=45,  cam_el=20, rot_x=0, rot_z=90),
    dict(name="grazing_sun",   sun_az=180, sun_el=8,  cam_az=45,  cam_el=20, rot_x=0, rot_z=0),
    dict(name="backlit",       sun_az=225, sun_el=15, cam_az=45,  cam_el=15, rot_x=0, rot_z=0),
]

SUN_STRENGTH = 4.0        # W/m^2 in Blender's units; 3-5 looks right with AgX/Filmic
SUN_ANGLE_DEG = 0.53      # angular size of the real Sun seen from Earth orbit
AMBIENT = 0.02            # faint fill ("Earthshine"); 0 = pitch-black shadows (realistic but harsh)
FRAMING = 1.0             # camera distance as a fraction of "just fits the bounding sphere"; <1 = tighter, may clip
RES = (1280, 720)
SAMPLES = 64


# --------------------------------------------------------------------------
# helpers
# --------------------------------------------------------------------------
def direction(az_deg, el_deg):
    """Unit vector for an azimuth/elevation pair (see conventions above)."""
    az, el = math.radians(az_deg), math.radians(el_deg)
    return Vector((math.cos(el) * math.cos(az), math.cos(el) * math.sin(az), math.sin(el)))


def look_along(obj, vec):
    """Orient obj so its -Z axis (camera / sun-light forward axis) points along vec."""
    obj.rotation_euler = vec.normalized().to_track_quat('-Z', 'Y').to_euler()


def world_bbox(meshes):
    pts = [o.matrix_world @ Vector(c) for o in meshes for c in o.bound_box]
    mn = Vector((min(p.x for p in pts), min(p.y for p in pts), min(p.z for p in pts)))
    mx = Vector((max(p.x for p in pts), max(p.y for p in pts), max(p.z for p in pts)))
    return mn, mx


def triangle_count(meshes):
    total = 0
    for o in meshes:
        o.data.calc_loop_triangles()
        total += len(o.data.loop_triangles)
    return total


def pick_engine(name):
    """Return the engine identifier that this Blender version accepts for 'EEVEE' / 'CYCLES'.

    Tries assigning each candidate instead of reading the engine enum, because the enum
    listing can leave out engines that come from add-ons (Cycles).
    """
    scene = bpy.context.scene
    original = scene.render.engine
    wanted = {'EEVEE': ['BLENDER_EEVEE', 'BLENDER_EEVEE_NEXT'], 'CYCLES': ['CYCLES']}[name.upper()]
    for cand in wanted:
        try:
            scene.render.engine = cand
            scene.render.engine = original
            return cand
        except (TypeError, ValueError):
            continue
    scene.render.engine = original
    raise RuntimeError(f"Render engine {name!r} is not available in this Blender "
                       "(for Cycles: Edit > Preferences > Add-ons > enable 'Cycles Render Engine').")


def setup_group(model):
    """Find the satellite, and give it a pivot empty at its centre so it rotates in place."""
    col = bpy.data.collections[model["collection"]]
    meshes = [o for o in col.all_objects if o.type == 'MESH']
    root = bpy.data.objects[model["root"]]
    pivot_name = model["root"].replace("_root", "_pivot")
    pivot = bpy.data.objects.get(pivot_name)
    if pivot is not None:
        pivot.rotation_euler = (0, 0, 0)          # measure the un-rotated model
    bpy.context.view_layer.update()
    mn, mx = world_bbox(meshes)
    radius = (mx - mn).length / 2
    centre = (mn + mx) / 2
    if pivot is None:
        pivot = bpy.data.objects.new(pivot_name, None)
        col.objects.link(pivot)
        pivot.location = centre
        bpy.context.view_layer.update()
        root.parent = pivot
        root.matrix_parent_inverse = pivot.matrix_world.inverted()
        bpy.context.view_layer.update()
    elif (pivot.matrix_world.translation - centre).length > 1e-4:
        # pivot has drifted away from the model's centre (e.g. the model was moved):
        # re-centre it without moving the model, so rotation and framing stay correct
        world = root.matrix_world.copy()
        pivot.location = centre
        bpy.context.view_layer.update()
        root.matrix_world = world
        bpy.context.view_layer.update()
    return dict(model=model, col=col, meshes=meshes, pivot=pivot, radius=radius,
                tris=triangle_count(meshes))


def ensure_sun():
    obj = bpy.data.objects.get("RenderSun")
    if obj is None:
        data = bpy.data.lights.new("RenderSun", 'SUN')
        obj = bpy.data.objects.new("RenderSun", data)
        bpy.context.scene.collection.objects.link(obj)
    obj.data.energy = SUN_STRENGTH
    obj.data.angle = math.radians(SUN_ANGLE_DEG)
    return obj


def ensure_camera():
    obj = bpy.data.objects.get("RenderCam")
    if obj is None:
        data = bpy.data.cameras.new("RenderCam")
        obj = bpy.data.objects.new("RenderCam", data)
        bpy.context.scene.collection.objects.link(obj)
    obj.data.lens = 50
    obj.data.clip_start = 0.1
    obj.data.clip_end = 5000
    return obj


def ensure_space_world(ambient=None):
    w = bpy.data.worlds.get("SSA_Space")
    if w is None:
        w = bpy.data.worlds.new("SSA_Space")
    w.use_nodes = True
    bg = w.node_tree.nodes.get("Background")
    bg.inputs["Color"].default_value = (0.5, 0.6, 1.0, 1.0)     # faint cool "Earthshine" tint
    bg.inputs["Strength"].default_value = AMBIENT if ambient is None else ambient                # 0 = pure black background/shadows
    return w


def configure_render(scene, engine, res, samples):
    scene.render.engine = pick_engine(engine)
    scene.render.resolution_x, scene.render.resolution_y = res
    scene.render.resolution_percentage = 100
    scene.render.image_settings.file_format = 'PNG'
    scene.render.film_transparent = False
    try:
        if scene.render.engine == 'CYCLES':
            scene.cycles.samples = samples
            scene.cycles.use_denoising = True
        else:
            scene.eevee.taa_render_samples = samples
    except Exception as exc:  # property names shift between Blender versions
        print("Could not set samples:", exc)


# --------------------------------------------------------------------------
# main
# --------------------------------------------------------------------------
def main(out_dir=None, engine="EEVEE", shots=None, models=None, res=RES, samples=SAMPLES,
         framing=None, filename=None, write_log=True, ambient=None):
    """filename(model_label, index, shot) -> str lets a caller choose output names (used for
    animation frames); write_log=False skips render_log.csv.
    ambient overrides AMBIENT (0 = pure black space, no fill light)."""
    scene = bpy.context.scene
    shots = SHOTS if shots is None else shots
    models = MODELS if models is None else models
    if out_dir is None:
        out_dir = bpy.path.abspath("//renders") if bpy.data.filepath else os.path.join(
            os.path.expanduser("~"), "ssa_renders")
    os.makedirs(out_dir, exist_ok=True)

    groups = [setup_group(m) for m in models]
    sun, cam, world = ensure_sun(), ensure_camera(), ensure_space_world(ambient)

    # remember everything we change so the scene is left as we found it
    saved = dict(camera=scene.camera, world=scene.world, engine=scene.render.engine,
                 res=(scene.render.resolution_x, scene.render.resolution_y),
                 path=scene.render.filepath,
                 # every satellite collection (not just the ones in this call) so the others can be hidden
                 col_hide={m["collection"]: bpy.data.collections[m["collection"]].hide_render
                           for m in MODELS if m["collection"] in bpy.data.collections},
                 other_lights={o.name: o.hide_render for o in bpy.data.objects
                               if o.type == 'LIGHT' and o is not sun},
                 pivot_rot={g["pivot"].name: tuple(g["pivot"].rotation_euler) for g in groups})
    log_rows = []
    try:
        configure_render(scene, engine, res, samples)
        scene.camera = cam
        scene.world = world
        for name in saved["other_lights"]:
            bpy.data.objects[name].hide_render = True       # only the Sun lights the scene

        for g in groups:
            for name in saved["col_hide"]:                  # show only the model being rendered,
                bpy.data.collections[name].hide_render = (name != g["col"].name)   # even if it is rendered alone
            centre = g["pivot"].matrix_world.translation.copy()
            # true limiting field of view for this picture shape (the camera's own angle_y
            # uses the sensor height and overestimates it for 16:9 renders)
            rx, ry = res
            half_long = math.tan(cam.data.angle_x / 2)   # sensor fit AUTO: 36 mm spans the longer side
            fov = 2 * math.atan(half_long * min(rx, ry) / max(rx, ry))
            fit = FRAMING if framing is None else framing
            dist = g["radius"] * fit / math.sin(fov / 2)        # fit < 1 = tighter (may clip); 1 = never clips

            for i, shot in enumerate(shots, 1):
                g["pivot"].rotation_euler = (math.radians(shot.get("rot_x", 0)), 0,
                                             math.radians(shot.get("rot_z", 0)))
                sun_dir = direction(shot["sun_az"], shot["sun_el"])      # centre -> Sun
                sun.location = centre + sun_dir * (dist * 2)
                look_along(sun, -sun_dir)                                 # light travels towards the satellite
                cam_dir = direction(shot["cam_az"], shot["cam_el"])
                cam.location = centre + cam_dir * dist
                look_along(cam, centre - cam.location)
                bpy.context.view_layer.update()

                fname = (filename(g['model']['label'], i, shot) if filename
                         else f"{g['model']['label']}_{i:02d}_{shot['name']}.png")
                scene.render.filepath = os.path.join(out_dir, fname)
                t0 = time.perf_counter()
                bpy.ops.render.render(write_still=True)
                secs = time.perf_counter() - t0
                print(f"[render] {fname}  {secs:.1f}s")
                log_rows.append([g["model"]["label"], g["tris"], shot["name"], shot["sun_az"],
                                 shot["sun_el"], shot["cam_az"], shot["cam_el"],
                                 shot.get("rot_x", 0), shot.get("rot_z", 0),
                                 scene.render.engine, f"{res[0]}x{res[1]}", samples,
                                 round(secs, 2), fname])
    finally:
        # put everything back
        scene.camera, scene.world = saved["camera"], saved["world"]
        scene.render.engine = saved["engine"]
        scene.render.resolution_x, scene.render.resolution_y = saved["res"]
        scene.render.filepath = saved["path"]
        for name, hide in saved["col_hide"].items():
            bpy.data.collections[name].hide_render = hide
        for name, hide in saved["other_lights"].items():
            bpy.data.objects[name].hide_render = hide
        for name, rot in saved["pivot_rot"].items():
            bpy.data.objects[name].rotation_euler = rot

    if write_log:
        log_path = os.path.join(out_dir, "render_log.csv")
        new = not os.path.exists(log_path)
        with open(log_path, "a", newline="") as fh:
            w = csv.writer(fh)
            if new:
                w.writerow(["model", "triangles", "shot", "sun_az", "sun_el", "cam_az", "cam_el",
                            "rot_x", "rot_z", "engine", "resolution", "samples", "seconds", "file"])
            w.writerows(log_rows)
    return dict(out_dir=out_dir, renders=len(log_rows),
                seconds=[(r[0], r[2], r[12]) for r in log_rows])


if __name__ == "__main__":
    import argparse
    argv = sys.argv[sys.argv.index("--") + 1:] if "--" in sys.argv else []
    p = argparse.ArgumentParser()
    p.add_argument("--out", default=None)
    p.add_argument("--engine", default="EEVEE", choices=["EEVEE", "CYCLES"])
    p.add_argument("--samples", type=int, default=SAMPLES)
    a = p.parse_args(argv)
    print(main(out_dir=a.out, engine=a.engine, samples=a.samples))

# FILE: dawn_dusk_scenario.py
################################################################################

"""
dawn_dusk_scenario.py - render a satellite on a dawn-dusk (terminator) polar orbit, spinning,
as seen by a ground observer at twilight.

It builds on render_angles.py (same folder): this script works out plausible Sun and
observer directions for every moment of one visible pass, then hands them to render_angles.main().

Run it:
  * Blender console / Scripting tab:
        import runpy
        ns = runpy.run_path(r"C:\\SSA\\FYProject_SSA_Training\\dawn_dusk_scenario.py", run_name="scenario")
        ns["run"]()                       # EEVEE, both satellites
        ns["run"](engine="CYCLES")        # slower, higher quality
  * Terminal:
        blender -b satellite_comparison.blend -P dawn_dusk_scenario.py -- --engine EEVEE

THE SCENARIO (simplified, not real orbit data)
  * Circular, sun-synchronous orbit at 705 km (Aqua's altitude), inclined ~98 deg, so it passes
    over the poles.  "Dawn-dusk" means the orbit plane is almost perpendicular to the Sun
    direction (beta angle ~85 deg): the satellite rides the day/night line and is sunlit all
    year, with the Sun low on its local horizon.
  * A ground observer sits slightly on the night side of the ground track, where the Sun is a few
    degrees below the horizon (twilight).  The sky is dark but the satellite is still in sunlight,
    which is the classic "bright satellite in the dusk sky" view.
  * The satellite spins about its own vertical (model Z) axis at a constant rate.
  * Earth rotation, atmosphere, Earth's shadow and the observer's real latitude are ignored.

LOCAL FRAME USED FOR THE ANGLES  (matches render_angles.py's az/el convention)
  +X = direction of flight, +Y = orbit normal (towards the Sun side), +Z = away from Earth.
  Azimuth is measured counter-clockwise from +X, elevation above the XY plane.
  So an observer on the ground appears at a NEGATIVE elevation (below the satellite).

ASSUMPTION about the models: model +Z is "up/zenith" and the satellite spins about it.
"""
import csv
import math
import os
import sys
import runpy

# ---------------------------------------------------------------------------
# scenario parameters - edit these
# ---------------------------------------------------------------------------
ALTITUDE_KM = 705.0         # Aqua's altitude (Suomi NPP is ~824 km)
BETA_DEG = 85.0             # angle between the Sun and the orbit plane; 90 = exactly dawn-dusk
OBSERVER_SUN_EL_DEG = -4.0  # Sun's elevation at the observer; -4 = just after sunset (twilight)
MIN_ELEVATION_DEG = 10.0    # satellite must be this high above the observer's horizon to be seen
SPIN_DEG_PER_S = 0.8        # satellite spin rate; 0.8 deg/s = one turn every 7.5 minutes
N_FRAMES = 10               # renders per satellite over the visible pass

R_EARTH = 6371.0            # km
MU = 398600.4418            # km^3/s^2


def _vec_sub(a, b): return tuple(x - y for x, y in zip(a, b))
def _dot(a, b): return sum(x * y for x, y in zip(a, b))
def _norm(a): return math.sqrt(_dot(a, a))


def compute_pass(n_frames=None):
    """Return (list of per-frame geometry dicts, metadata dict) for n_frames points along the pass."""
    n_frames = N_FRAMES if n_frames is None else n_frames
    r = R_EARTH + ALTITUDE_KM
    period = 2 * math.pi * math.sqrt(r ** 3 / MU)            # seconds
    rate = 360.0 / period                                     # orbital deg per second
    beta = math.radians(BETA_DEG)
    # sun direction in the orbit-fixed frame (x = orbit normal, y = towards closest approach)
    sun = (math.sin(beta), math.cos(beta), 0.0)
    # observer sits on the night side so the Sun is at OBSERVER_SUN_EL_DEG there:
    # cos(beta - delta) = sin(sun_el)  ->  delta = beta - 90 + sun_el
    delta = math.radians(BETA_DEG - 90.0 + OBSERVER_SUN_EL_DEG)
    obs = (R_EARTH * math.sin(delta), R_EARTH * math.cos(delta), 0.0)
    obs_up = tuple(c / R_EARTH for c in obs)

    samples = []
    g = -45.0
    while g <= 45.0:                                          # orbital phase, deg from closest approach
        gr = math.radians(g)
        pos = (0.0, r * math.cos(gr), r * math.sin(gr))
        to_sat = _vec_sub(pos, obs)
        rng = _norm(to_sat)
        elev = math.degrees(math.asin(_dot(to_sat, obs_up) / rng))
        if elev >= MIN_ELEVATION_DEG:
            samples.append((g, elev, rng))
        g += 0.05
    if len(samples) < 2:
        raise RuntimeError("No visible pass with these settings - lower MIN_ELEVATION_DEG or move the observer.")

    g_start, g_end = samples[0][0], samples[-1][0]
    frames = []
    for i in range(n_frames):
        g = g_start + (g_end - g_start) * i / (n_frames - 1)
        gr = math.radians(g)
        pos = (0.0, r * math.cos(gr), r * math.sin(gr))
        # local axes at the satellite: forward v, normal c (= +x), up u
        v = (0.0, -math.sin(gr), math.cos(gr))
        c = (1.0, 0.0, 0.0)
        u = (0.0, math.cos(gr), math.sin(gr))
        to_obs = _vec_sub(obs, pos)
        rng = _norm(to_obs)
        elev = math.degrees(math.asin(_dot(_vec_sub(pos, obs), obs_up) / rng))

        def to_az_el(vec):
            local = (_dot(vec, v), _dot(vec, c), _dot(vec, u))
            n = _norm(local)
            return (math.degrees(math.atan2(local[1], local[0])) % 360.0,
                    math.degrees(math.asin(local[2] / n)))

        sun_az, sun_el = to_az_el(sun)
        cam_az, cam_el = to_az_el(to_obs)
        t = (g - g_start) / rate                              # seconds since the pass began
        # phase angle = angle Sun-satellite-observer (0 = Sun behind the observer, 180 = backlit)
        phase = math.degrees(math.acos(max(-1.0, min(1.0, _dot(sun, to_obs) / rng))))
        frames.append(dict(i=i + 1, t=t, orbit_deg=g, obs_elev=elev, range_km=rng, phase=phase,
                           sun_az=sun_az, sun_el=sun_el, cam_az=cam_az, cam_el=cam_el,
                           spin=(SPIN_DEG_PER_S * t) % 360.0))
    meta = dict(period_min=period / 60, pass_seconds=(g_end - g_start) / rate,
                max_elev=max(s[1] for s in samples), observer_offset_km=abs(R_EARTH * delta),
                spin_turns=SPIN_DEG_PER_S * (g_end - g_start) / rate / 360.0)
    return frames, meta

"""
make_animation.py - animate the dawn-dusk pass as a side-by-side video (detailed vs low-poly).

  ns = runpy.run_path(r"C:\\SSA\\FYProject_SSA_Training\\make_animation.py", run_name="anim")
  ns["render_frames"](max_seconds=100)   # repeat until it reports remaining == 0 (resumable)
  ns["assemble"]()                      # builds the MP4 from the rendered frames

Every frame uses the same geometry as dawn_dusk_scenario.py (Sun, observer and spin for that
moment of the pass), just sampled densely.  The 8.3-minute pass is compressed into SECONDS of video.
"""
import os
import time
import runpy
import bpy

FPS = 24
SECONDS = 10                       # video length; the 497 s pass is played back faster than real time
N = FPS * SECONDS
RES = (1280, 720)                  # per satellite; the video is twice as wide
FRAMES_DIR = r"C:\SSA\animation_frames_v2"   # outside the repo (hundreds of MB of PNGs)
VIDEO = r"C:\SSA\FYProject_SSA_Training\dawn_dusk_pass.mp4"
FRAMING = 1.05
LABELS = {"Aqua_detailed": "Aqua  -  detailed (115k triangles)",
          "SuomiNPP_lowpoly": "Suomi NPP  -  low-poly (4k triangles)"}

_here = os.path.dirname(os.path.abspath(globals().get("__file__", "")))
if not os.path.exists(os.path.join(_here, "render_angles.py")):
    _here = r"C:\SSA\FYProject_SSA_Training"


def _load():
    ra = runpy.run_path(os.path.join(_here, "render_angles.py"), run_name="render_angles")
    sc = runpy.run_path(os.path.join(_here, "dawn_dusk_scenario.py"), run_name="scenario")
    frames, meta = sc["compute_pass"](N)
    return ra, frames, meta


def _frame_path(label, i):
    return os.path.join(FRAMES_DIR, label, f"frame_{i:04d}.png")


def render_frames(max_seconds=100, engine="EEVEE", samples=64, batch=8):
    """Render whichever frames are still missing, stopping after about max_seconds."""
    t0 = time.perf_counter()
    ra, frames, meta = _load()
    done_now = 0
    for model in ra["MODELS"]:
        label = model["label"]
        os.makedirs(os.path.join(FRAMES_DIR, label), exist_ok=True)
        todo = [f for f in frames if not os.path.exists(_frame_path(label, f["i"]))]
        for k in range(0, len(todo), batch):
            if time.perf_counter() - t0 > max_seconds:
                break
            chunk = todo[k:k + batch]
            shots = [dict(name=f"frame_{f['i']:04d}", sun_az=f["sun_az"], sun_el=f["sun_el"],
                          cam_az=f["cam_az"], cam_el=f["cam_el"], rot_x=0, rot_z=f["spin"])
                     for f in chunk]
            ra["main"](out_dir=os.path.join(FRAMES_DIR, label), engine=engine, shots=shots,
                       models=[model], res=RES, samples=samples, framing=FRAMING,
                       filename=lambda lbl, i, shot: shot["name"] + ".png", write_log=False)
            done_now += len(chunk)
    remaining = sum(1 for m in ra["MODELS"] for f in frames
                    if not os.path.exists(_frame_path(m["label"], f["i"])))
    return dict(rendered_this_call=done_now, remaining=remaining, total=N * len(ra["MODELS"]),
                seconds=round(time.perf_counter() - t0, 1))


def assemble(video=VIDEO):
    """Join the rendered frames into one side-by-side MP4 using Blender's video sequencer."""
    ra, frames, meta = _load()
    labels = [m["label"] for m in ra["MODELS"]]
    missing = [(l, f["i"]) for l in labels for f in frames if not os.path.exists(_frame_path(l, f["i"]))]
    if missing:
        raise RuntimeError(f"{len(missing)} frames not rendered yet (first missing: {missing[0]})")

    name = "SSA_Video"
    if name in bpy.data.scenes:
        bpy.data.scenes.remove(bpy.data.scenes[name])
    sc = bpy.data.scenes.new(name)
    sc.render.resolution_x, sc.render.resolution_y = RES[0] * 2, RES[1]
    sc.render.resolution_percentage = 100
    sc.render.fps = FPS
    sc.frame_start, sc.frame_end = 1, N
    se = sc.sequence_editor_create()
    strips = se.strips if hasattr(se, "strips") else se.sequences

    for ch, (label, off) in enumerate(zip(labels, (-RES[0] / 2, RES[0] / 2)), start=1):
        first = _frame_path(label, 1)
        s = strips.new_image(label, filepath=first, channel=ch, frame_start=1)
        for i in range(2, N + 1):
            s.elements.append(os.path.basename(_frame_path(label, i)))
        s.transform.offset_x = off
        s.transform.offset_y = 0

    def text(strip_name, channel, start, end, body, x, y, size, color):
        t = strips.new_effect(name=strip_name, type='TEXT', channel=channel,
                              frame_start=start, length=end - start)
        t.text, t.font_size, t.color = body, size, color
        t.location = (x, y)
        t.use_shadow = True
        return t

    text("label_left", 4, 1, N + 1, LABELS[labels[0]], 0.25, 0.94, 30, (1, 1, 1, 1))
    text("label_right", 5, 1, N + 1, LABELS[labels[1]], 0.75, 0.94, 30, (1, 1, 1, 1))
    for f in frames:
        body = (f"pass time {f['t']:3.0f} s   |   observer elevation {f['obs_elev']:2.0f} deg   |   "
                f"Sun phase angle {f['phase']:3.0f} deg   |   spin {f['spin']:3.0f} deg")
        text(f"info_{f['i']}", 6, f["i"], f["i"] + 1, body, 0.5, 0.05, 24, (1, 0.92, 0.6, 1))

    r = sc.render
    r.use_sequencer = True
    if hasattr(r.image_settings, "media_type"):      # Blender 5+: pick "video" before the FFMPEG format
        r.image_settings.media_type = 'VIDEO'
    r.image_settings.file_format = 'FFMPEG'
    r.ffmpeg.format = 'MPEG4'
    r.ffmpeg.codec = 'H264'
    r.ffmpeg.constant_rate_factor = 'MEDIUM'
    r.filepath = video
    os.makedirs(os.path.dirname(video), exist_ok=True)
    bpy.ops.render.render(animation=True, scene=name)
    bpy.data.scenes.remove(sc)
    # Blender may add a frame range to the name (e.g. dawn_dusk_pass0001-0240.mp4): find what it wrote
    folder, stem = os.path.dirname(video), os.path.splitext(os.path.basename(video))[0]
    written = sorted(f for f in os.listdir(folder) if f.startswith(stem) and f.endswith(".mp4"))
    paths = [os.path.join(folder, f) for f in written]
    return dict(videos=paths, size_mb=[round(os.path.getsize(p) / 1e6, 2) for p in paths],
                frames=N, fps=FPS, seconds=SECONDS)

"""
material_tests.py - how brightness depends on angle for three kinds of surface.

Runs in its own temporary scene, so your satellite scene is not touched.
Usage (Blender console):

def _material(name, metallic, roughness):
    m = bpy.data.materials.get(name) or bpy.data.materials.new(name)
    m.use_nodes = True
    bsdf = m.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = (0.8, 0.8, 0.8, 1.0)
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    if "Specular IOR Level" in bsdf.inputs:           # matte surfaces: no extra specular highlight
        bsdf.inputs["Specular IOR Level"].default_value = 0.0 if metallic == 0 else 0.5
    return m


def _plate_mesh(name):
    bm = bmesh.new()
    bmesh.ops.create_grid(bm, x_segments=1, y_segments=1, size=PLATE / 2)
    bm.transform(Matrix.Rotation(math.radians(90), 4, 'X'))      # face the camera (normal -Y)
    me = bpy.data.meshes.new(name)
    bm.to_mesh(me)
    bm.free()
    return me


def _sphere_mesh(name):
    bm = bmesh.new()
    bmesh.ops.create_uvsphere(bm, u_segments=96, v_segments=48, radius=SPHERE_R)
    me = bpy.data.meshes.new(name)
    bm.to_mesh(me)
    bm.free()
    for p in me.polygons:
        p.use_smooth = True
    return me


def _clean_old():

def build_scene():

             "sphere_diffuse": (_sphere_mesh("MT_sphere"), mats["matte"]),
             "panel_shiny": (_plate_mesh("MT_plate_s"), mats["mirror"])}
    for key, (mesh, mat) in specs.items():
        ob = bpy.data.objects.new("MT_" + key, mesh)
        ob.data.materials.append(mat)
        col.objects.link(ob)

    cam_data = bpy.data.cameras.new("MT_cam")

    col.objects.link(cam)
    sc.camera = cam

    sun_data = bpy.data.lights.new("MT_sun", 'SUN')

    sun_data.angle = math.radians(0.53)

    bg.inputs["Strength"].default_value = 0.0
    sc.world = w

    r = sc.render
    for cand in ("BLENDER_EEVEE", "BLENDER_EEVEE_NEXT"):

    sc.view_settings.look = 'None'

def _render_and_measure(sc, tmp_path, png_path=None):
    """Render, then return (total linear brightness, apparent area in pixels, render seconds)."""
    sc.render.image_settings.file_format = 'OPEN_EXR'
    sc.render.filepath = tmp_path
    t0 = time.perf_counter()
    bpy.ops.render.render(scene=sc.name, write_still=True)
    secs = time.perf_counter() - t0
    if png_path:                                          # keep a viewable picture
        sc.render.image_settings.file_format = 'PNG'
        sc.render.image_settings.color_depth = '8'
        bpy.data.images["Render Result"].save_render(png_path, scene=sc)
        sc.render.image_settings.file_format = 'OPEN_EXR'
        sc.render.image_settings.color_depth = '32'
    exr = tmp_path + ".exr"
    img = bpy.data.images.load(exr)
    img.colorspace_settings.name = 'Non-Color'
    px = np.empty(len(img.pixels), dtype=np.float32)
    img.pixels.foreach_get(px)
    bpy.data.images.remove(img)
    os.remove(exr)
    px = px.reshape(-1, 4)
    lum = 0.2126 * px[:, 0] + 0.7152 * px[:, 1] + 0.0722 * px[:, 2]
    return float(lum.sum()), float(px[:, 3].sum()), secs


def _angles(key):
    coarse = [a * STEP_DEG for a in range(int(360 / STEP_DEG))]
    if key != "panel_shiny":
        return sorted(coarse)
    # a mirror flashes only over a fraction of a degree of spin, so sample very finely there:
    # 0.1 deg steps within +-3 deg and 0.5 deg steps within +-12 deg of each predicted flash
    extra = []
    for centre in (PHASE_DEG / 2, PHASE_DEG / 2 + 180):
        extra += [centre + k * 0.5 for k in range(-24, 25)]
        extra += [centre + k * 0.1 for k in range(-30, 31)]
    allv = {round(a % 360, 4) for a in coarse + extra}
    return sorted(allv)


def spin_sweep(key):
    """Spin one object through 360 degrees; write material_tests/data/spin_<key>.csv."""
    assert key in OBJECTS, key
    os.makedirs(os.path.join(OUT_DIR, "data"), exist_ok=True)
    os.makedirs(os.path.join(OUT_DIR, "samples"), exist_ok=True)
    sc = build_scene()
    _show_only(sc, key)
    _ortho_camera(sc)
    ob = bpy.data.objects["MT_" + key]
    tmp = os.path.join(OUT_DIR, "data", "_tmp_" + key)
    rows = []
    t0 = time.perf_counter()
    try:
        for ang in _angles(key):
            ob.rotation_euler = (0, 0, math.radians(ang))
            bpy.context.view_layer.update()
            png = None
            if any(abs(ang - s) < 1e-6 for s in SAMPLE_ANGLES):
                png = os.path.join(OUT_DIR, "samples", f"{key}_{int(round(ang)):03d}.png")
            flux, area, secs = _render_and_measure(sc, tmp, png)
            rows.append((key, round(ang, 4), flux, area))
    finally:
        bpy.data.scenes.remove(sc)
    path = os.path.join(OUT_DIR, "data", f"spin_{key}.csv")
    with open(path, "w", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["object", "spin_deg", "brightness", "apparent_area_px"])
        w.writerows(rows)
    return dict(object=key, frames=len(rows), seconds=round(time.perf_counter() - t0, 1), csv=path,
                max_brightness=max(r[2] for r in rows))


def distance_sweep():
    """Matte sphere at increasing distance from a perspective camera."""
    os.makedirs(os.path.join(OUT_DIR, "data"), exist_ok=True)
    sc = build_scene()
    _show_only(sc, "sphere_diffuse")
    tmp = os.path.join(OUT_DIR, "data", "_tmp_dist")
    rows = []
    try:
        for d in DISTANCES:
            _persp_camera(sc, d)
            bpy.context.view_layer.update()
            flux, area, _ = _render_and_measure(sc, tmp)
            rows.append(("sphere_diffuse", d, flux, area))
    finally:
        bpy.data.scenes.remove(sc)
    path = os.path.join(OUT_DIR, "data", "distance_sphere_diffuse.csv")
    with open(path, "w", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["object", "distance_m", "brightness", "apparent_area_px"])
        w.writerows(rows)
    return dict(csv=path, rows=len(rows))

"""
fidelity_test.py - how much does model detail matter for brightness at orbital distance?

VARIANTS (all rendered at identical Sun / camera / spin angles)
  A        Aqua detailed (your 115k-triangle model) - the reference
  N4k      Aqua, plain automatic decimation to ~4k triangles (shrinks thin flat parts: a BAD cheap model)
  D30k/D10k/D5k/D4k  Aqua reduced to ~30k / 10k / 5k / 4k triangles with the thin flat parts (solar
                   arrays, panels, antennas) kept whole and only the bulky body parts simplified
  C        Suomi NPP low-poly (your supplied low-poly model - a different satellite, so shape differs too)

WHAT IS MEASURED
  * brightness  = total linear light reaching the camera (sum of pixel luminance, 32-bit EXR)
  * the camera is ORTHOGRAPHIC: at orbital range the rays from the satellite are parallel
  * washout: the picture is summed into blocks of k x k pixels (k = 1, 2, 4 ... 512).  Big blocks mimic a
    distant object / coarse sensor.  E(k) = sum|A_k - B_k| / sum(A_k) is the fractional brightness
    difference between detailed and decimated that survives at that resolution.  k = 512 is the
    whole picture in one pixel, i.e. a point source: E = |flux_B - flux_A| / flux_A.

GEOMETRIES
  spin sweeps  : the satellite spins about its vertical (Z) axis, Sun and camera fixed in the satellite
                 frame, at Sun-camera phase angles 30, 90 and 150 deg (5 deg steps)
  dawn-dusk    : the 48-frame pass from dawn_dusk_scenario.py (Sun, observer and spin change together)

Usage (Blender console):
  ns = runpy.run_path(r"C:\\SSA\\FYProject_SSA_Training\\fidelity_test.py", run_name="fid")
  ns["run"]()            # resumable: finished geometries are skipped
"""
import csv
import math
import os
import runpy
import time

import bmesh
import bpy
import numpy as np
from mathutils import Matrix, Vector

HERE = r"C:\SSA\FYProject_SSA_Training"
OUT_DIR = os.path.join(HERE, "fidelity_test")
RES = 512
SAMPLES = 16
SUN_STRENGTH = 4.0
SPIN_STEP = 5
PHASES = (30, 90, 150)
CAM_AZ, CAM_EL = 0.0, 20.0
PASS_FRAMES = 48
LEVELS = [2 ** i for i in range(10)]          # block sizes 1 ... 512
TARGET_TRIS = 4024                            # the supplied low-poly model's triangle count
SCENE = "SSA_Fidelity"

MODELS = {"A": dict(collection="Satellite_Detailed", root="Aqua_Detailed_root"),
          "C": dict(collection="Satellite_LowPoly", root="SuomiNPP_LowPoly_root")}


def direction(az, el):
    a, e = math.radians(az), math.radians(el)
    return Vector((math.cos(e) * math.cos(a), math.cos(e) * math.sin(a), math.sin(e)))


def look_along(obj, vec):
    obj.rotation_euler = vec.normalized().to_track_quat('-Z', 'Y').to_euler()


def tri_count(objs_or_meshes):
    n = 0
    for o in objs_or_meshes:
        me = o.data if hasattr(o, "data") else o
        me.calc_loop_triangles()
        n += len(me.loop_triangles)
    return n


def _co(me):
    n = len(me.vertices)
    co = np.empty(n * 3, dtype=np.float32)
    me.vertices.foreach_get("co", co)
    return co.reshape(n, 3).astype(np.float64)


def world_bbox(pairs):
    pts = []
    for me, mw in pairs:
        if len(me.vertices):
            m = np.array(mw)
            pts.append(_co(me) @ m[:3, :3].T + m[:3, 3])
    p = np.concatenate(pts)
    return Vector(p.min(axis=0)), Vector(p.max(axis=0))


def world_area(pairs):
    tot = 0.0
    for me, mw in pairs:
        me.calc_loop_triangles()
        nt = len(me.loop_triangles)
        if nt == 0:
            continue
        m = np.array(mw)
        w = _co(me) @ m[:3, :3].T + m[:3, 3]
        t = np.empty(nt * 3, dtype=np.int32)
        me.loop_triangles.foreach_get("vertices", t)
        t = t.reshape(-1, 3)
        tot += 0.5 * np.linalg.norm(np.cross(w[t[:, 1]] - w[t[:, 0]], w[t[:, 2]] - w[t[:, 0]]), axis=1).sum()
    return float(tot)

KEEP = 800          # parts with at most this many triangles (flat panels, arrays, antennas) are kept whole


def _source_meshes(key):
    m = MODELS[key]
    pivot = bpy.data.objects.get(m["root"].replace("_root", "_pivot"))
    old_rot = tuple(pivot.rotation_euler) if pivot else None
    if pivot:
        pivot.rotation_euler = (0, 0, 0)
    bpy.context.view_layer.update()
    col = bpy.data.collections[m["collection"]]
    objs = [o for o in col.all_objects if o.type == 'MESH' and not o.hide_render]
    return [(o, o.matrix_world.copy()) for o in objs], (pivot, old_rot)


class _Reducer:
    """Temporary copies of the source parts with a planar-merge and/or collapse modifier."""

    def __init__(self, pairs):
        self.pairs = pairs
        self.col = bpy.data.collections.new("FID_tmp")
        bpy.context.scene.collection.children.link(self.col)
        self.copies = []
        for o, mw in pairs:
            c = o.copy()
            c.parent = None
            c.matrix_world = mw
            self.col.objects.link(c)
            self.copies.append(c)

    def evaluate(self, planar, ratios):
        """planar: merge flat faces first; ratios: per-part collapse ratio (None = no collapse)."""
        for c, r in zip(self.copies, ratios):
            for md in list(c.modifiers):
                c.modifiers.remove(md)
            if planar:
                d = c.modifiers.new("planar", 'DECIMATE')
                d.decimate_type = 'DISSOLVE'
                d.angle_limit = math.radians(2.0)
                d.delimit = {'MATERIAL'}
            if r is not None and r < 1.0:
                d = c.modifiers.new("collapse", 'DECIMATE')
                d.decimate_type = 'COLLAPSE'
                d.ratio = r
                d.use_collapse_triangulate = True
        bpy.context.view_layer.update()
        dg = bpy.context.evaluated_depsgraph_get()
        return [bpy.data.meshes.new_from_object(c.evaluated_get(dg), depsgraph=dg) for c in self.copies]

    def close(self):
        for c in self.copies:
            bpy.data.objects.remove(c)
        bpy.data.collections.remove(self.col)


def _reduce(red, planar, target, protect):
    """Return (meshes, total triangles).  target=None: planar merge only."""
    none = [None] * len(red.copies)
    if target is None:
        ms = red.evaluate(planar, none)
        return ms, tri_count(ms)
    if not protect:                                           # naive: one ratio for everything
        n0 = tri_count([o for o, _ in red.pairs])
        r = target / n0
        best = None
        for _ in range(4):
            ms = red.evaluate(False, [r] * len(none))
            tot = tri_count(ms)
            if best:
                [bpy.data.meshes.remove(m) for m in best[0]]
            best = (ms, tot)
            if abs(tot - target) / target < 0.05:
                break
            r = min(1.0, r * target / max(1, tot))
        return best
    base = red.evaluate(planar, none)
    n = [tri_count([m]) for m in base]
    [bpy.data.meshes.remove(m) for m in base]
    prot = sum(x for x in n if x <= KEEP)
    big = sum(x for x in n if x > KEEP)
    r = max(0.001, min(1.0, (target - prot) / max(1, big)))
    best = None
    for _ in range(5):
        ms = red.evaluate(planar, [r if x > KEEP else None for x in n])
        tot = tri_count(ms)
        if best:
            [bpy.data.meshes.remove(m) for m in best[0]]
        best = (ms, tot)
        if abs(tot - target) / target < 0.05:
            break
        got_big = sum(tri_count([m]) for m, x in zip(ms, n) if x > KEEP)
        r = max(0.001, min(1.0, r * (target - prot) / max(1, got_big)))
    return best


VARIANT_ORDER = ["A", "N4k", "D30k", "D10k", "D5k", "D4k", "C"]
PICTURE_VARIANTS = ["A", "N4k", "D4k"]


def build_scene():
    if SCENE in bpy.data.scenes:
        bpy.data.scenes.remove(bpy.data.scenes[SCENE])
    sc = bpy.data.scenes.new(SCENE)
    restore = []
    srcA, rA = _source_meshes("A")
    srcC, rC = _source_meshes("C")
    restore += [rA, rC]
    sets = {"A": [(o.data, mw) for o, mw in srcA], "C": [(o.data, mw) for o, mw in srcC]}
    red = _Reducer(srcA)
    try:
        for name, planar, target, protect in (("N4k", False, TARGET_TRIS, False), ("D30k", False, 30000, True),
                                     ("D10k", False, 10000, True), ("D5k", False, 5000, True),
                                     ("D4k", False, TARGET_TRIS, True)):
            ms, tot = _reduce(red, planar, target, protect)
            sets[name] = [(m, mw) for m, (_, mw) in zip(ms, srcA)]
    finally:
        red.close()

    info = {}
    centres, radii = {}, {}
    for k in ("A", "C"):
        mn, mx = world_bbox(sets[k])
        centres[k], radii[k] = (mn + mx) / 2, (mx - mn).length / 2
        info["extent_" + k] = tuple(round(v, 3) for v in (mx - mn))
    areaA = world_area(sets["A"])
    V = {}
    for k in VARIANT_ORDER:
        base = "C" if k == "C" else "A"
        col = bpy.data.collections.new("FID_" + k)
        sc.collection.children.link(col)
        spin = bpy.data.objects.new("FID_spin_" + k, None)
        col.objects.link(spin)
        for i, (me, mw) in enumerate(sets[k]):
            ob = bpy.data.objects.new(f"FID_{k}_{i}", me)
            ob.matrix_world = Matrix.Translation(-centres[base]) @ mw
            col.objects.link(ob)
            ob.parent = spin
            ob.matrix_parent_inverse = Matrix.Identity(4)
        V[k] = dict(col=col, spin=spin, radius=radii[base], tris=tri_count([m for m, _ in sets[k]]),
                    area=world_area(sets[k]))
    info["variants"] = {k: dict(tris=V[k]["tris"], area=round(V[k]["area"], 3),
                                area_vs_A=round(V[k]["area"] / areaA, 4)) for k in V if k != "C"}
    info["variants"]["C"] = dict(tris=V["C"]["tris"], area=round(V["C"]["area"], 3))

    cam = bpy.data.objects.new("FID_cam", bpy.data.cameras.new("FID_cam"))
    sc.collection.objects.link(cam)
    sc.camera = cam
    cam.data.type = 'ORTHO'
    cam.data.clip_start, cam.data.clip_end = 0.1, 1000
    sun = bpy.data.objects.new("FID_sun", bpy.data.lights.new("FID_sun", 'SUN'))
    sc.collection.objects.link(sun)
    sun.data.energy = SUN_STRENGTH
    sun.data.angle = math.radians(0.53)

    w = bpy.data.worlds.get("FID_black") or bpy.data.worlds.new("FID_black")
    w.use_nodes = True
    w.node_tree.nodes["Background"].inputs["Color"].default_value = (0, 0, 0, 1)
    w.node_tree.nodes["Background"].inputs["Strength"].default_value = 0.0
    sc.world = w

    r = sc.render
    for cand in ("BLENDER_EEVEE", "BLENDER_EEVEE_NEXT"):
        try:
            r.engine = cand
            break
        except TypeError:
            continue
    try:
        sc.eevee.taa_render_samples = SAMPLES
    except Exception:
        pass
    r.resolution_x = r.resolution_y = RES
    r.resolution_percentage = 100
    r.film_transparent = True
    sc.view_settings.view_transform = 'Standard'
    sc.view_settings.look = 'None'
    r.image_settings.file_format = 'OPEN_EXR'
    r.image_settings.color_depth = '32'
    return sc, V, cam, sun, restore, info


def cleanup(restore):
    for pivot, rot in restore:
        if pivot is not None and rot is not None:
            pivot.rotation_euler = rot
    if SCENE in bpy.data.scenes:
        sc = bpy.data.scenes[SCENE]
        objs = [o for c in sc.collection.children_recursive for o in c.objects] + list(sc.collection.objects)
        bpy.data.scenes.remove(sc)
        for o in set(objs):
            bpy.data.objects.remove(o)
        for c in [c for c in bpy.data.collections if c.name.startswith("FID_")]:
            bpy.data.collections.remove(c)
    for me in [m for m in bpy.data.meshes if m.users == 0]:
        bpy.data.meshes.remove(me)

def render_lum(sc, tmp):
    sc.render.filepath = tmp
    bpy.ops.render.render(scene=sc.name, write_still=True)
    exr = tmp + ".exr"
    img = bpy.data.images.load(exr)
    img.colorspace_settings.name = 'Non-Color'
    w, h = img.size
    px = np.empty(len(img.pixels), dtype=np.float32)
    img.pixels.foreach_get(px)
    bpy.data.images.remove(img)
    os.remove(exr)
    px = px.reshape(h, w, 4)
    lum = 0.2126 * px[..., 0] + 0.7152 * px[..., 1] + 0.0722 * px[..., 2]
    a = px[..., 3]
    edge = float(a[0].sum() + a[-1].sum() + a[:, 0].sum() + a[:, -1].sum())
    return lum.astype(np.float64), edge


def binned(L, k):
    n = L.shape[0] // k
    return L.reshape(n, k, n, k).sum(axis=(1, 3))


def errors(A, B):
    out = []
    for k in LEVELS:
        a, b = binned(A, k), binned(B, k)
        s = a.sum()
        out.append(float(np.abs(a - b).sum() / s) if s > 0 else float("nan"))
    return out


def configs():
    cams = {}
    for ph in PHASES:
        cams[f"spin_phase{ph:03d}"] = [
            dict(sun_az=CAM_AZ + ph, sun_el=CAM_EL, cam_az=CAM_AZ, cam_el=CAM_EL, spin=s, t=None)
            for s in range(0, 360, SPIN_STEP)]
    ns = runpy.run_path(os.path.join(HERE, "dawn_dusk_scenario.py"), run_name="scenario")
    frames, meta = ns["compute_pass"](PASS_FRAMES)
    cams["dawn_dusk_pass"] = [dict(sun_az=f["sun_az"], sun_el=f["sun_el"], cam_az=f["cam_az"],
                                   cam_el=f["cam_el"], spin=f["spin"], t=f["t"]) for f in frames]
    return cams


def do_config(sc, V, cam, sun, name, geoms):
    os.makedirs(os.path.join(OUT_DIR, "data"), exist_ok=True)
    tmp = os.path.join(OUT_DIR, "data", "_tmp")
    rows, keep, secs = [], {}, {k: 0.0 for k in VARIANT_ORDER}
    for i, g in enumerate(geoms):
        sd, cd = direction(g["sun_az"], g["sun_el"]), direction(g["cam_az"], g["cam_el"])
        phase = math.degrees(math.acos(max(-1, min(1, sd.dot(cd)))))
        sun.location = sd * 100
        look_along(sun, -sd)
        cam.location = cd * 100
        look_along(cam, -cd)
        lum = {}
        for k in VARIANT_ORDER:
            for kk in VARIANT_ORDER:
                V[kk]["col"].hide_render = (kk != k)
                V[kk]["spin"].rotation_euler = (0, 0, math.radians(g["spin"]))
            cam.data.ortho_scale = 2.0 * V[k]["radius"] * 1.03
            bpy.context.view_layer.update()
            t0 = time.perf_counter()
            lum[k], edge = render_lum(sc, tmp)
            secs[k] += time.perf_counter() - t0
            if k == "A":
                edgeA = edge
        fluxA = lum["A"].sum()
        for k in VARIANT_ORDER:
            E = errors(lum["A"], lum[k]) if k not in ("A", "C") else [float("nan")] * len(LEVELS)
            rows.append([name, i, round(g["spin"], 3), round(phase, 2), k, V[k]["tris"], fluxA,
                         lum[k].sum(), edgeA] + E)
        keep[i] = (fluxA, {k: lum[k] for k in PICTURE_VARIANTS})
    order = sorted(keep, key=lambda j: keep[j][0])
    pick = [order[-1], order[int(len(order) * 0.5)]]
    arrays = {f"{k}_{n}": np.stack([keep[j][1][k] for j in pick]).astype(np.float16)
              for n, k in enumerate(PICTURE_VARIANTS)}
    np.savez_compressed(os.path.join(OUT_DIR, "data", f"pictures_{name}.npz"), idx=np.array(pick),
                        names=np.array(PICTURE_VARIANTS), **arrays)
    path = os.path.join(OUT_DIR, "data", f"frames_{name}.csv")
    with open(path, "w", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["config", "frame", "spin_deg", "phase_deg", "variant", "triangles", "flux_A_reference",
                    "flux", "border_alpha_A"] + [f"E_block{k}" for k in LEVELS])
        w.writerows(rows)
    return dict(config=name, frames=len(geoms), seconds={k: round(v, 1) for k, v in secs.items()})


def run(only=None, force=False):
    import json
    cfgs = configs()
    names = [n for n in cfgs if (only is None or n in only)]
    todo = [n for n in names if force or not os.path.exists(os.path.join(OUT_DIR, "data", f"frames_{n}.csv"))]
    if not todo:
        return "all done"
    sc, V, cam, sun, restore, info = build_scene()
    info.update(res=RES, samples=SAMPLES)
    os.makedirs(os.path.join(OUT_DIR, "data"), exist_ok=True)
    with open(os.path.join(OUT_DIR, "data", "run_info.json"), "w") as fh:
        json.dump(info, fh, indent=1)
    done = []
    try:
        for n in todo:
            done.append(do_config(sc, V, cam, sun, n, cfgs[n]))
            with open(os.path.join(OUT_DIR, "data", "run_info_done.json"), "w") as fh:
                json.dump(done, fh, indent=1)
    finally:
        cleanup(restore)
    return dict(info, done=done)

"""
sun_sweep.py - fixed satellite, fixed camera, a Sun that travels round the satellite on an ellipse.

The Sun starts directly ABOVE the satellite, swings down one side, passes UNDER it, and comes back up the
other side to the top.  Nothing else moves: the satellite does not spin and the camera does not move, so
every change in the pictures is due to the lighting alone.

THE PATH (all directions are seen from the satellite; +Z is up)
  The Sun sits on an ellipse in a vertical plane:
      position(t) = A * sin(t) * h  +  B * cos(t) * z          t = 0 ... 360 deg
  with h a horizontal axis, z straight up, A the horizontal and B the vertical radius of the ellipse.
  t = 0 is straight overhead, t = 90 is level on the h side, t = 180 is straight underneath, t = 270 is
  level on the other side.  A > B stretches the ellipse sideways, so the Sun moves faster over the top and
  bottom than near the horizon.  Only the Sun's direction matters for lighting (it is very far away).
  The plane is turned ELLIPSE_AZ_OFFSET degrees away from the camera direction so the Sun passes in front
  of the satellite on one side and behind it on the other.

Run (Blender console / Scripting tab):
  ns = runpy.run_path(r"C:\\SSA\\FYProject_SSA_Training\\sun_sweep.py", run_name="sweep")
  ns["run"]()                 # both satellites, 36 positions each (every 10 degrees along the ellipse)
Builds on render_angles.py (same folder).
"""
import csv
import math
import os
import runpy

from mathutils import Vector

HERE = r"C:\SSA\FYProject_SSA_Training"

CAM_AZ, CAM_EL = 45.0, 10.0         # fixed camera direction as seen from the satellite
ELLIPSE_AZ_OFFSET = 60.0            # horizontal axis of the ellipse, degrees away from the camera azimuth
A, B = 1.6, 1.0                     # horizontal and vertical radii of the ellipse (only their ratio matters)
STEP_DEG = 10                       # step along the ellipse (360 / STEP = number of pictures)
RES = (1280, 720)
SAMPLES = 64
AMBIENT = 0.0                       # 0 = pure black space: surfaces facing away from the Sun are completely dark


def sun_path(step=STEP_DEG):
    """List of dicts: ellipse angle t, Sun azimuth/elevation (render_angles convention), phase angle."""
    az_h = math.radians(CAM_AZ + ELLIPSE_AZ_OFFSET)
    h = Vector((math.cos(az_h), math.sin(az_h), 0.0))
    z = Vector((0.0, 0.0, 1.0))
    cam = Vector((math.cos(math.radians(CAM_EL)) * math.cos(math.radians(CAM_AZ)),
                  math.cos(math.radians(CAM_EL)) * math.sin(math.radians(CAM_AZ)),
                  math.sin(math.radians(CAM_EL))))
    out = []
    for i in range(int(round(360 / step))):
        t = i * step
        p = A * math.sin(math.radians(t)) * h + B * math.cos(math.radians(t)) * z
        d = p.normalized()
        out.append(dict(i=i + 1, t=t,
                        sun_az=math.degrees(math.atan2(d.y, d.x)) % 360.0,
                        sun_el=math.degrees(math.asin(max(-1.0, min(1.0, d.z)))),
                        phase=math.degrees(math.acos(max(-1.0, min(1.0, d.dot(cam)))))))
    return out


def run(out_dir=None, engine="EEVEE", samples=SAMPLES, res=RES, models=None, framing=1.05, ambient=AMBIENT):
    ra = runpy.run_path(os.path.join(HERE, "render_angles.py"), run_name="render_angles")
    out_dir = out_dir or os.path.join(HERE, "renders_sun_sweep")
    path = sun_path()
    shots = [dict(name=f"sun{int(p['t']):03d}", sun_az=p["sun_az"], sun_el=p["sun_el"],
                  cam_az=CAM_AZ, cam_el=CAM_EL, rot_x=0, rot_z=0) for p in path]
    result = ra["main"](out_dir=out_dir, engine=engine, shots=shots, models=models, res=res,
                        samples=samples, framing=framing, write_log=False, ambient=ambient,
                        filename=lambda lbl, i, shot: f"{lbl}_{shot['name']}.png")
    os.makedirs(out_dir, exist_ok=True)
    with open(os.path.join(out_dir, "sun_path.csv"), "w", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["frame", "ellipse_deg", "sun_az", "sun_el", "phase_deg"])
        for p in path:
            w.writerow([p["i"], p["t"], round(p["sun_az"], 2), round(p["sun_el"], 2), round(p["phase"], 2)])
    return dict(result, frames=len(path))


if __name__ == "__main__":
    print(run())

"""
spin_animation.py - render the frames for the "spin, Sun 90 deg" animation (fidelity test, top-right graph).

The satellite spins 360 degrees about its vertical axis; the Sun and the camera stay fixed.
Three versions of Aqua are shown at identical angles:
    A    detailed (115k triangles)
    D4k  cheap, shape kept (~4.8k triangles)
    N4k  plain automatic reduction (4k triangles)
It reuses the scene builder from fidelity_test.py (same models, same Sun, same orthographic camera), so the
pictures match the brightness numbers in fidelity_test/data/frames_spin_phase090.csv.

Run (Blender console / Scripting tab):
  ns = runpy.run_path(r"C:\\SSA\\FYProject_SSA_Training\\spin_animation.py", run_name="spin")
  ns["run"]()         # resumable: frames that already exist are skipped
Then run make_spin_animation.py (any Python with Pillow, matplotlib, numpy, ffmpeg) to build the video.
"""
import math
import os
import runpy

import bpy

HERE = r"C:\SSA\FYProject_SSA_Training"
OUT = os.path.join(HERE, "fidelity_test", "animation_spin90")
VARIANTS = ["A", "D4k", "N4k"]
PHASE = 90                      # Sun azimuth offset from the camera (same as the fidelity test's "spin, 90")
STEP = 5                        # degrees of spin per frame (72 frames)
RES = 720
SAMPLES = 32


def frame_path(variant, spin):
    return os.path.join(OUT, f"{variant}_{int(spin):03d}.png")


def run(max_seconds=900):
    import time
    t0 = time.perf_counter()
    spins = list(range(0, 360, STEP))
    todo = [(v, s) for s in spins for v in VARIANTS if not os.path.exists(frame_path(v, s))]
    if not todo:
        return dict(done=True, remaining=0)
    os.makedirs(OUT, exist_ok=True)
    ft = runpy.run_path(os.path.join(HERE, "fidelity_test.py"), run_name="fid")
    sc, V, cam, sun, restore, info = ft["build_scene"]()
    sc.render.resolution_x = sc.render.resolution_y = RES
    try:
        sc.eevee.taa_render_samples = SAMPLES
    except Exception:
        pass
    sc.render.image_settings.file_format = 'PNG'
    sc.render.image_settings.color_depth = '8'
    sc.render.film_transparent = False
    cam_az, cam_el = ft["CAM_AZ"], ft["CAM_EL"]
    sd, cd = ft["direction"](cam_az + PHASE, cam_el), ft["direction"](cam_az, cam_el)
    sun.location = sd * 100
    ft["look_along"](sun, -sd)
    cam.location = cd * 100
    ft["look_along"](cam, -cd)
    done = 0
    try:
        for s in spins:
            for v in VARIANTS:
                if not os.path.exists(frame_path(v, s)):
                    if time.perf_counter() - t0 > max_seconds:
                        raise StopIteration
                    for k in V:
                        V[k]["col"].hide_render = (k != v)
                        V[k]["spin"].rotation_euler = (0, 0, math.radians(s))
                    cam.data.ortho_scale = 2.0 * V["A"]["radius"] * 1.03      # same scale for all three
                    bpy.context.view_layer.update()
                    sc.render.filepath = frame_path(v, s)
                    bpy.ops.render.render(scene=sc.name, write_still=True)
                    done += 1
    except StopIteration:
        pass
    finally:
        ft["cleanup"](restore)
    remaining = sum(1 for s in spins for v in VARIANTS if not os.path.exists(frame_path(v, s)))
    return dict(rendered_this_call=done, remaining=remaining, total=len(spins) * len(VARIANTS),
                seconds=round(time.perf_counter() - t0, 1))

def stats():
    """Numbers that back up the graphs (written to summary_stats.json)."""
    D = os.path.join(ROOT, "material_tests", "data") + os.sep
    def load(n):
        a = np.genfromtxt(D + n, delimiter=",", names=True, dtype=None, encoding="utf-8")
        return a
    plate, sph, shiny, dist = (load(n) for n in ("spin_plate_diffuse.csv", "spin_sphere_diffuse.csv",
                                                 "spin_panel_shiny.csv", "distance_sphere_diffuse.csv"))
    ref = sph["brightness"].mean()
    out = {"reference_matte_sphere_flux": float(ref), "sphere_spread_pct": float(100 * (sph["brightness"].max() - sph["brightness"].min()) / ref)}

    # plate: Lambert theory  flux ~ k * [cos(i)+ * cos(e)+] for front and back faces
    th = np.radians(plate["spin_deg"]); c1 = np.cos(th - np.radians(60)); c2 = np.cos(th)
    model = np.maximum(c1, 0) * np.maximum(c2, 0) + np.maximum(-c1, 0) * np.maximum(-c2, 0)
    k = (model * plate["brightness"]).sum() / (model * model).sum()
    rel = plate["brightness"] / ref
    out["plate"] = {"peak_rel": float(rel.max()), "peak_at_deg": float(plate["spin_deg"][rel.argmax()]),
                    "theory_rms_err_pct_of_peak": float(100 * np.sqrt(np.mean((plate["brightness"] - k * model) ** 2)) / plate["brightness"].max()),
                    "frac_of_spin_above_50pct_of_peak": float(np.mean(plate["brightness"] > 0.5 * plate["brightness"].max())),
                    "frac_of_spin_dark_below_1pct": float(np.mean(plate["brightness"] < 0.01 * plate["brightness"].max())),
                    "area_vs_cos_max_dev_pct": float(100 * np.max(np.abs(plate["apparent_area_px"] / plate["apparent_area_px"].max() - np.abs(np.cos(th)))))}

    # shiny
    a = shiny["spin_deg"]; f = shiny["brightness"]
    peak = f.max(); out["shiny"] = {"peak_rel": float(peak / ref), "peak_at_deg": float(a[f.argmax()])}
    floor = np.median(f); out["shiny"]["floor_rel"] = float(floor / ref)
    out["shiny"]["peak_over_floor"] = float(peak / floor)
    for centre in (30, 210):
        m = np.abs(a - centre) <= 3; aa, ff = a[m], f[m]
        half = ff.max() / 2; above = aa[ff >= half]
        out["shiny"][f"flash_{centre}"] = {"peak_rel": float(ff.max() / ref), "peak_at": float(aa[ff.argmax()]),
                                           "fwhm_deg": float(above.max() - above.min()), "n_points_in_fwhm": int(len(above))}
    # fraction of the 360 deg during which it is brighter than the matte sphere (reference)
    # (coarse 2 deg grid only, so the fine sampling does not bias the fraction)
    coarse = np.isclose(a % 2, 0) | np.isclose(a % 2, 2)
    out["shiny"]["coarse_frac_brighter_than_sphere"] = float(np.mean(f[coarse] > ref))
    out["shiny"]["fraction_of_spin_above_1pct_of_peak_estimate_deg"] = float(
        sum(np.diff(np.sort(a[f > 0.01 * peak]))[np.diff(np.sort(a[f > 0.01 * peak])) < 1.0]))

    # distance
    d = dist["distance_m"].astype(float)
    sl_b = np.polyfit(np.log(d), np.log(dist["brightness"]), 1)[0]
    sl_a = np.polyfit(np.log(d), np.log(dist["apparent_area_px"]), 1)[0]
    out["distance"] = {"brightness_slope": float(sl_b), "area_slope": float(sl_a),
                       "brightness_over_area_spread_pct": float(100 * np.ptp(dist["brightness"] / dist["apparent_area_px"]) / np.mean(dist["brightness"] / dist["apparent_area_px"]))}
    print(json.dumps(out, indent=1))
    json.dump(out, open(os.path.join(D, "summary_stats.json"), "w"), indent=1)
    return out

"""
plot_fidelity.py - the graphs for the detailed-vs-cheap model comparison (fidelity test).

  python plot_fidelity.py [repo folder]        (default: the folder above this script)

Reads   fidelity_test/data/frames_*.csv, pictures_*.npz, run_info.json   (made by fidelity_test.py in Blender)
Writes  fidelity_test/data/summary_stats.json   errors per model and geometry (the numbers quoted)
        fidelity_test/fidelity_main.png          the 2 x 2 figure: brightness curves, error vs triangles, washout
        fidelity_test/fidelity_pictures.png      the same frame at coarser and coarser pixel blocks
Needs Python with numpy and matplotlib.  Run it outside Blender (any Python).

Model names used in the data:  A = detailed Aqua (115k), N4k = plain automatic reduction (4k),
D30k / D10k / D5k / D4k = shape-kept reductions, C = Suomi NPP low-poly (a different satellite).
"""
import csv

import math

def _write(dest, saver):
    """Save via a temp file, then copy over: writing straight onto an existing file can fail on some mounted drives."""
    tmp = os.path.join(tempfile.gettempdir(), "_plot_tmp_" + os.path.basename(dest))
    saver(tmp)
    shutil.copyfile(tmp, dest)
    os.remove(tmp)


def stats():
    """Error of each cheap model against the detailed one, for each geometry (written to summary_stats.json)."""
    D=os.path.join(ROOT,"fidelity_test","data")+os.sep
    rows=[]
    for f in sorted(glob.glob(D+"frames_*.csv")):
        rows+=list(csv.DictReader(open(f)))
    cfgs=sorted({r["config"] for r in rows})
    vars_=["A","N4k","D30k","D10k","D5k","D4k","C"]
    info=json.load(open(D+"run_info.json"))
    def get(cfg,v,key="flux"):
        rs=[r for r in rows if r["config"]==cfg and r["variant"]==v]
        rs.sort(key=lambda r:int(r["frame"]))
        return np.array([float(r[key]) for r in rs])
    LV=[2**i for i in range(10)]
    out={}
    for cfg in cfgs:
        fa=get(cfg,"A","flux")
        o={"fluxA_max":fa.max(),"fluxA_mean":fa.mean(),"frac_dark":float((fa<0.01*fa.max()).mean())}
        for v in vars_[1:]:
            fv=get(cfg,v,"flux")
            d={}
            d["weighted_err"]=float(np.abs(fv-fa).sum()/fa.sum())
            d["bias"]=float(fv.sum()/fa.sum()-1)
            lit=fa>0.05*fa.max()
            d["max_err_lit"]=float(np.max(np.abs(fv-fa)[lit]/fa[lit]))
            d["r"]=float(np.corrcoef(fa,fv)[0,1])
            # shape error after best constant rescale
            s=(fa*fv).sum()/(fv*fv).sum()
            d["shape_err"]=float(np.abs(s*fv-fa).sum()/fa.sum())
            d["scale"]=float(s)
            if v!="C":
                E=[]
                for k in LV:
                    e=get(cfg,v,f"E_block{k}")
                    E.append(float((e*fa).sum()/fa.sum()))
                d["E"]=E
            o[v]=d
        out[cfg]=o
    json.dump(out,open(D+"summary_stats.json","w"),indent=1)
    for cfg in cfgs:
        print(cfg, "dark frac %.2f"%out[cfg]["frac_dark"])
        for v in vars_[1:]:
            d=out[cfg][v]; print("  %-5s werr %.3f bias %+.3f maxlit %.3f r %.4f shape %.3f"%(v,d["weighted_err"],d["bias"],d["max_err_lit"],d["r"],d["shape_err"]), "E1 %.3f E32 %.3f E128 %.3f E512 %.3f"%(d["E"][0],d["E"][5],d["E"][7],d["E"][9]) if "E" in d else "")
    print(info["variants"])
    return out

def pictures():

    z=np.load(D+"pictures_spin_phase090.npz")

    A=z["A_0"][0].astype(float); NB=z["N4k_1"][0].astype(float); DB=z["D4k_2"][0].astype(float)
    rows=[r for r in csv.DictReader(open(D+"frames_spin_phase090.csv")) if int(r["frame"])==idx]

    ys,xs=np.where(A>0); y0,y1,x0,x1=ys.min(),ys.max()+1,xs.min(),xs.max()+1
